cmake_minimum_required(VERSION 3.24)
project(YTChatterino VERSION 2.5.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Svg WebSockets)
add_library(chatterino_core STATIC
    src/core.cpp src/core.hpp src/network.cpp src/network.hpp
    src/kick.cpp src/kickauth.cpp
    src/accounts.cpp src/accounts.hpp src/chat.cpp src/chat.hpp src/automod.cpp src/automod.hpp
    src/view.cpp src/view.hpp src/application.cpp src/application.hpp src/settings.cpp
    src/vod.cpp src/vod.hpp src/phone.cpp src/phone.hpp src/vendor/qrcodegen.cpp src/vendor/qrcodegen.hpp
    src/updater.cpp src/updater.hpp)
target_link_libraries(chatterino_core PUBLIC Qt6::Widgets Qt6::Network Qt6::Svg Qt6::WebSockets)
target_include_directories(chatterino_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(chatterino_core PUBLIC APP_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_sources(chatterino_core PRIVATE src/obsupdate.cpp src/obsupdate.hpp)
  target_link_libraries(chatterino_core PUBLIC advapi32 crypt32 winmm dwmapi shell32)
endif()
if(MSVC)
  target_compile_options(chatterino_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(chatterino_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(BUILD_STANDALONE "Build the standalone chat application" ON)
if(BUILD_STANDALONE)
  qt_add_executable(YT-Chatterino WIN32 src/main.cpp assets/resources.qrc)
  target_link_libraries(YT-Chatterino PRIVATE chatterino_core)
endif()
option(BUILD_OBS_LINUX "Build the OBS dock for Linux" OFF)
if(BUILD_OBS_LINUX AND UNIX AND NOT APPLE)
  find_package(libobs REQUIRED)
  find_package(obs-frontend-api REQUIRED)
  add_library(yt-chatterino-obs MODULE obs-plugin/src/linux-plugin.cpp assets/resources.qrc)
  target_link_libraries(yt-chatterino-obs PRIVATE chatterino_core OBS::libobs OBS::obs-frontend-api)
  set_target_properties(yt-chatterino-obs PROPERTIES PREFIX "" OUTPUT_NAME "yt-chatterino-obs")
  install(TARGETS yt-chatterino-obs LIBRARY DESTINATION lib/obs-plugins)
  install(DIRECTORY browser-extension/ DESTINATION share/yt-chatterino/browser-extension)
  add_executable(obs-linux-fixture obs-plugin/tests/linux-host.cpp)
  target_link_libraries(obs-linux-fixture PRIVATE Qt6::Widgets ${CMAKE_DL_LIBS})
  target_link_options(obs-linux-fixture PRIVATE -Wl,--export-dynamic)
endif()
if(BUILD_STANDALONE AND WIN32)
  enable_language(RC)
  configure_file(packaging/windows/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
  target_sources(YT-Chatterino PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
elseif(BUILD_STANDALONE)
  set_target_properties(YT-Chatterino PROPERTIES OUTPUT_NAME yt-chatterino)
endif()
include(CTest)
option(BUILD_BENCHMARKS "Build the isolated desktop benchmark" OFF)
if(BUILD_BENCHMARKS)
  qt_add_executable(chatterino_benchmark benchmarks/native.cpp assets/resources.qrc)
  target_link_libraries(chatterino_benchmark PRIVATE chatterino_core)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
  qt_add_executable(chatterino_tests tests/tests.cpp tests/extra.cpp tests/v210.cpp tests/v211.cpp tests/v212.cpp tests/v213.cpp tests/v214.cpp tests/v220.cpp tests/v221.cpp tests/v222.cpp tests/v223.cpp tests/v224.cpp tests/v225.cpp tests/v226.cpp tests/v230.cpp tests/v231.cpp tests/v232.cpp tests/v233.cpp tests/v240.cpp tests/v241.cpp tests/v242.cpp tests/v243.cpp tests/v244.cpp tests/v245.cpp tests/v246.cpp tests/v247.cpp tests/v248.cpp tests/v249.cpp tests/v250.cpp tests/v251.cpp tests/v252.cpp tests/v253.cpp tests/v254.cpp tests/v255.cpp tests/v256.cpp tests/native_tests.hpp assets/resources.qrc)
  target_sources(chatterino_tests PRIVATE tests/v257.cpp)
  target_link_libraries(chatterino_tests PRIVATE chatterino_core Qt6::Test)
  add_test(NAME native_features COMMAND chatterino_tests)
  set_tests_properties(native_features PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(BUILD_STANDALONE)
  install(TARGETS YT-Chatterino RUNTIME DESTINATION bin)
endif()
if(BUILD_STANDALONE AND UNIX AND NOT APPLE)
  install(FILES packaging/flatpak/io.github.GuiCraftBR_eu.YTChatterino.desktop DESTINATION share/applications)
  install(FILES assets/ytchatterino-logo.png DESTINATION share/icons/hicolor/256x256/apps RENAME io.github.GuiCraftBR_eu.YTChatterino.png)
  install(FILES packaging/flatpak/io.github.GuiCraftBR_eu.YTChatterino.metainfo.xml DESTINATION share/metainfo)
  install(DIRECTORY browser-extension/ DESTINATION share/yt-chatterino/browser-extension)
  install(DIRECTORY docs/ DESTINATION share/doc/yt-chatterino)
  install(DIRECTORY Licenses/ DESTINATION share/licenses/yt-chatterino)
endif()
