cmake_minimum_required(VERSION 3.24)
project(YTChatterino VERSION 2.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Svg)
add_library(chatterino_core STATIC
    src/core.cpp src/core.hpp src/network.cpp src/network.hpp
    src/accounts.cpp src/accounts.hpp src/chat.cpp src/chat.hpp
    src/view.cpp src/view.hpp src/application.cpp src/application.hpp src/settings.cpp
    src/updater.cpp src/updater.hpp)
target_link_libraries(chatterino_core PUBLIC Qt6::Widgets Qt6::Network Qt6::Svg)
target_include_directories(chatterino_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(chatterino_core PUBLIC APP_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_link_libraries(chatterino_core PUBLIC advapi32 crypt32 winmm dwmapi shell32)
endif()
if(MSVC)
  target_compile_options(chatterino_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(chatterino_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(YT-Chatterino WIN32 src/main.cpp assets/resources.qrc)
target_link_libraries(YT-Chatterino PRIVATE chatterino_core)
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
  target_sources(YT-Chatterino PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
else()
  set_target_properties(YT-Chatterino PROPERTIES OUTPUT_NAME yt-chatterino)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
  qt_add_executable(chatterino_tests tests/tests.cpp tests/extra.cpp tests/native_tests.hpp assets/resources.qrc)
  target_link_libraries(chatterino_tests PRIVATE chatterino_core Qt6::Test)
  add_test(NAME native_features COMMAND chatterino_tests)
  set_tests_properties(native_features PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS YT-Chatterino RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(FILES packaging/arch/yt-chatterino.desktop DESTINATION share/applications)
  install(FILES assets/ytchatterino-logo.png DESTINATION share/icons/hicolor/256x256/apps RENAME yt-chatterino.png)
  install(DIRECTORY docs/ DESTINATION share/doc/yt-chatterino)
  install(DIRECTORY Licenses/ DESTINATION share/licenses/yt-chatterino)
endif()
