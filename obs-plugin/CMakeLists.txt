cmake_minimum_required(VERSION 3.24)
project(YTChatterinoObs VERSION 2.5.10 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT WIN32)
  message(FATAL_ERROR "The OBS dock host currently requires Windows.")
endif()
option(BUILD_OBS_DOCK_TESTS "Build an isolated OBS frontend fixture" OFF)
if(BUILD_OBS_DOCK_TESTS)
  add_library(obs-frontend-fixture SHARED tests/fake-frontend.cpp)
  target_link_libraries(obs-frontend-fixture PRIVATE Qt6::Widgets)
  set_target_properties(obs-frontend-fixture PROPERTIES PREFIX "" OUTPUT_NAME "obs-frontend-api")
  add_executable(obs-dock-fixture tests/host.cpp)
  target_link_libraries(obs-dock-fixture PRIVATE Qt6::Widgets user32)
  add_library(yt-chatterino-obs-test-bootstrap MODULE src/bootstrap.cpp)
  target_link_libraries(yt-chatterino-obs-test-bootstrap PRIVATE Qt6::Core user32)
  target_compile_definitions(yt-chatterino-obs-test-bootstrap PRIVATE YT_OBS_VERSION="${PROJECT_VERSION}" YT_OBS_BOOTSTRAP_TEST=1)
  set_target_properties(yt-chatterino-obs-test-bootstrap PROPERTIES PREFIX "" OUTPUT_NAME "yt-chatterino-obs-test-bootstrap")
endif()
add_library(yt-chatterino-obs MODULE src/bootstrap.cpp)
add_library(yt-chatterino-obs-core MODULE src/plugin.cpp)
target_link_libraries(yt-chatterino-obs PRIVATE Qt6::Core user32)
target_link_libraries(yt-chatterino-obs-core PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets user32)
target_compile_definitions(yt-chatterino-obs PRIVATE YT_OBS_VERSION="${PROJECT_VERSION}")
set_target_properties(yt-chatterino-obs PROPERTIES PREFIX "" OUTPUT_NAME "yt-chatterino-obs")
set_target_properties(yt-chatterino-obs-core PROPERTIES PREFIX "" OUTPUT_NAME "yt-chatterino-obs-core")
if(MSVC)
  target_compile_options(yt-chatterino-obs PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(yt-chatterino-obs-core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(yt-chatterino-obs PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(yt-chatterino-obs-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
