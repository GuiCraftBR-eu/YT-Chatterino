cmake_minimum_required(VERSION 3.24)
project(YTChatterinoObs VERSION 2.5.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT WIN32)
  message(FATAL_ERROR "The OBS dock host currently requires Windows.")
endif()
option(BUILD_OBS_DOCK_TESTS "Build an isolated OBS frontend fixture" OFF)
if(BUILD_OBS_DOCK_TESTS)
  add_library(obs-frontend-fixture SHARED tests/fake-frontend.cpp)
  target_link_libraries(obs-frontend-fixture PRIVATE Qt6::Widgets)
  set_target_properties(obs-frontend-fixture PROPERTIES PREFIX "" OUTPUT_NAME "obs-frontend-api")
  add_executable(obs-dock-fixture tests/host.cpp)
  target_link_libraries(obs-dock-fixture PRIVATE Qt6::Widgets user32)
endif()
add_library(yt-chatterino-obs MODULE src/plugin.cpp)
target_link_libraries(yt-chatterino-obs PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets user32)
set_target_properties(yt-chatterino-obs PROPERTIES PREFIX "" OUTPUT_NAME "yt-chatterino-obs")
if(MSVC)
  target_compile_options(yt-chatterino-obs PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(yt-chatterino-obs PRIVATE -Wall -Wextra -Wpedantic)
endif()
